$url = 'https://sentinel-ai-46hj.onrender.com/'
$deadline = (Get-Date).AddMinutes(6)
while ((Get-Date) -lt $deadline) {
    try {
        $r = Invoke-WebRequest -Uri $url -UseBasicParsing -TimeoutSec 60
        "HTTP $($r.StatusCode)"
        $r.Content
        break
    } catch {
        "HTTP $($_.Exception.Response.StatusCode.value__) - esperando despliegue..."
        Start-Sleep -Seconds 15
    }
}
